    \draw[thick,blue,samples=400,domain=0:10]

    \draw[thick,red,samples=200,domain=0:10]

    \draw[<->] (2,-1.05) -- (3,-1.05) node[midway,below] {$T=2\pi/\omega_0$};

    \node[red,anchor=west] at (4.2,0.8) {honey: $\propto e^{-(2-\sqrt{3})\omega_0t}$, rate $\approx3.3\times10^{7}\,\mathrm{s^{-1}}$};
